cmake_minimum_required(VERSION 3.20)
project(DDCSwitcherWindows VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(ddc_core src/core.cpp)
target_include_directories(ddc_core PUBLIC include)
if(MSVC)
  target_compile_options(ddc_core PUBLIC /utf-8 /W4 /EHsc)
else()
  target_compile_options(ddc_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(ddc_core_tests tests/core_tests.cpp)
target_link_libraries(ddc_core_tests PRIVATE ddc_core)
enable_testing()
add_test(NAME core COMMAND ddc_core_tests)
if(WIN32)
  enable_language(RC)
  add_executable(DDCSwitcher WIN32 src/app.cpp src/platform.cpp tests/platform_tests.cpp resources/app.rc)
  target_compile_definitions(DDCSwitcher PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_include_directories(DDCSwitcher PRIVATE resources)
  target_link_libraries(DDCSwitcher PRIVATE ddc_core dxva2 user32 shell32 ole32 uuid comctl32 gdi32)
  if(MINGW)
    target_link_options(DDCSwitcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  if(MSVC)
    target_link_options(DDCSwitcher PRIVATE /MANIFEST:NO)
  endif()
endif()
